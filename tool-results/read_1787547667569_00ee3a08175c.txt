    16→export function fmtDate(d) {
    17→  if(!d) return '';
    18→  try {
    19→    var ds = String(d).trim();
    20→    if (ds.includes('T')) { /* ISO ya tiene T */ }
    21→    else if (ds.includes(' ')) { ds = ds.replace(' ', 'T'); } /* SQLite: 2025-01-15 12:30:00 → 2025-01-15T12:30:00 */
    22→    else { ds = ds + 'T00:00:00'; } /* solo fecha: 2025-01-15 */
    23→    const dt = new Date(ds);
    24→    if (isNaN(dt.getTime())) return d;
    25→    return dt.getDate()+' '+['Ene','Feb','Mar','Abr','May','Jun','Jul','Ago','Sep','Oct','Nov','Dic'][dt.getMonth()]+' '+dt.getFullYear();
    26→  } catch(e) { return d; }
    27→}

    29→export function getVideoEmbed(url) {
    30→  if (!url) return '';
    31→  const ytMatch = url.match(/(?:youtube\.com\/watch\?v=|youtu\.be\/)([a-zA-Z0-9_-]+)/);
    32→  if (ytMatch) return `<iframe src="https://www.youtube.com/embed/${ytMatch[1]}" frameborder="0" allowfullscreen></iframe>`;
    33→  const ytShortsMatch = url.match(/youtube\.com\/shorts\/([a-zA-Z0-9_-]+)/);
    34→  if (ytShortsMatch) return `<iframe src="https://www.youtube.com/embed/${ytShortsMatch[1]}" frameborder="0" allowfullscreen></iframe>`;
    35→  const ttMatch = url.match(/tiktok\.com\/@[^/]+\/video\/(\d+)/);
    36→  if (ttMatch) return `<iframe data-tiktok src="https://www.tiktok.com/embed/v2/${ttMatch[1]}" frameborder="0" allowfullscreen></iframe>`;
    37→  if (/\.(mp4|webm|ogg|mov)(\?|$)/i.test(url)) {
    38→    return `<video src="${esc(url)}" controls playsinline preload="metadata" style="width:100%;border-radius:12px;background:#000;"></video>`;
    39→  }
    40→  return `<iframe src="${esc(url)}" frameborder="0" allowfullscreen></iframe>`;
    41→}
