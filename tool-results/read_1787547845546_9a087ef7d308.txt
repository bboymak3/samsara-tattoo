  1282→function normalizeSocialUrl(value, platform) {
  1283→  if (!value) return '';
  1284→  const v = value.trim();
  1285→  if (v.startsWith('http://') || v.startsWith('https://')) return v;
  1286→  let clean = v.replace(/^@/, '');
  1287→  // Strip common prefixes users type by mistake: "facebook.com/", "instagram.com/", "facebook/", etc.
  1288→  clean = clean.replace(/^(facebook\.com\/?|www\.facebook\.com\/?|facebook\/)/i, '');
  1289→  clean = clean.replace(/^(instagram\.com\/?|www\.instagram\.com\/?|instagram\/)/i, '');
  1290→  clean = clean.replace(/^(x\.com\/?|twitter\.com\/?|www\.twitter\.com\/?|twitter\/)/i, '');
  1291→  clean = clean.replace(/^(tiktok\.com\/?|www\.tiktok\.com\/?|tiktok\/)/i, '');
  1292→  clean = clean.replace(/^(youtube\.com\/?|www\.youtube\.com\/?|youtube\/)/i, '');
  1293→  clean = clean.replace(/^@/, ''); // Remove @ again if it appeared after stripping
  1294→  switch (platform) {
  1295→    case 'instagram': return `https://www.instagram.com/${clean}`;
  1296→    case 'facebook': return `https://www.facebook.com/${clean}`;
  1297→    case 'twitter': return `https://x.com/${clean}`;
  1298→    case 'tiktok': return `https://www.tiktok.com/@${clean}`;
  1299→    case 'youtube': return `https://www.youtube.com/@${clean}`;
  1300→    default: return v;
  1301→  }
  1302→}

  1304→function getVideoEmbed(url) {
  1305→  if (!url) return '';
  1306→  // Direct video files (R2 uploads): .mp4, .webm, .ogg, .mov — these are lightweight, load normally
  1307→  if (/\.(mp4|webm|ogg|mov)(\?|$)/i.test(url)) {
  1308→    return `<div style="border-radius:16px;overflow:hidden;box-shadow:0 4px 20px rgba(0,0,0,0.1);"><video src="${escapeHtml(url)}" controls playsinline preload="metadata" style="width:100%;border-radius:16px;background:#000;"></video></div>`;
  1309→  }
  1310→  // YouTube
  1311→  const ytMatch = url.match(/(?:youtube\.com\/watch\?v=|youtu\.be\/)([a-zA-Z0-9_-]+)/);
  1312→  if (ytMatch) return getThumbnailEmbed(`https://www.youtube.com/embed/${ytMatch[1]}`, `https://img.youtube.com/vi/${ytMatch[1]}/hqdefault.jpg`, 'Ver video en YouTube');
  1313→  // YouTube Shorts
  1314→  const ytShortsMatch = url.match(/youtube\.com\/shorts\/([a-zA-Z0-9_-]+)/);
  1315→  if (ytShortsMatch) return getThumbnailEmbed(`https://www.youtube.com/embed/${ytShortsMatch[1]}`, `https://img.youtube.com/vi/${ytShortsMatch[1]}/hqdefault.jpg`, 'Ver video en YouTube');
  1316→  // TikTok
  1317→  const ttMatch = url.match(/tiktok\.com\/@[^/]+\/video\/(\d+)/);
  1318→  if (ttMatch) return getLazyIframe(`https://www.tiktok.com/embed/v2/${ttMatch[1]}`, 'Ver video en TikTok');
  1319→  // Generic iframe
  1320→  return getLazyIframe(url, 'Reproducir video');
  1321→}
