     4→/**
     5→ * Escape a string for safe HTML attribute / text interpolation.
     6→ */
     7→export function escapeHtml(str) {
     8→  if (!str) return '';
     9→  return str
    10→    .replace(/&/g, '&amp;')
    11→    .replace(/</g, '&lt;')
    12→    .replace(/>/g, '&gt;')
    13→    .replace(/"/g, '&quot;')
    14→    .replace(/'/g, '&#39;');
    15→}
    16→
    17→/**
    18→ * Escape a string for safe JS single-quoted string interpolation.
    19→ */
    20→export function escapeJs(str) {
    21→  if (!str) return '';
    22→  return str.replace(/\\/g, '\\\\').replace(/'/g, "\\'").replace(/\n/g, '');
    23→}

    25→/**
    26→ * Normalize a social-media handle or partial URL into a full URL.
    27→ */
    28→export function normalizeSocialUrl(value, platform) {
    29→  if (!value) return '';
    30→  const v = value.trim();
    31→  if (v.startsWith('http://') || v.startsWith('https://')) return v;
    32→  let clean = v.replace(/^@/, '');
    33→  clean = clean.replace(/^(facebook\.com\/?|www\.facebook\.com\/?|facebook\/)/i, '');
    34→  clean = clean.replace(/^(instagram\.com\/?|www\.instagram\.com\/?|instagram\/)/i, '');
    35→  clean = clean.replace(/^(x\.com\/?|twitter\.com\/?|www\.twitter\.com\/?|twitter\/)/i, '');
    36→  clean = clean.replace(/^(tiktok\.com\/?|www\.tiktok\.com\/?|tiktok\/)/i, '');
    37→  clean = clean.replace(/^(youtube\.com\/?|www\.youtube\.com\/?|youtube\/)/i, '');
    38→  clean = clean.replace(/^@/, '');
    39→  switch (platform) {
    40→    case 'instagram': return `https://www.instagram.com/${clean}`;
    41→    case 'facebook': return `https://www.facebook.com/${clean}`;
    42→    case 'twitter': return `https://x.com/${clean}`;
    43→    case 'tiktok': return `https://www.tiktok.com/@${clean}`;
    44→    case 'youtube': return `https://www.youtube.com/@${clean}`;
    45→    default: return v;
    46→  }
    47→}

    49→/**
    50→ * Return an <iframe> or <video> embed string for a given video URL.
    51→ */
    52→export function getVideoEmbed(url) {
    53→  if (!url) return '';
    54→  // YouTube: youtube.com/watch?v=XXX or youtu.be/XXX
    55→  const ytMatch = url.match(/(?:youtube\.com\/watch\?v=|youtu\.be\/)([a-zA-Z0-9_-]+)/);
    56→  if (ytMatch) return `<iframe src="https://www.youtube.com/embed/${ytMatch[1]}" width="100%" style="aspect-ratio:9/16;" frameborder="0" allowfullscreen></iframe>`;
    57→  // YouTube Shorts: youtube.com/shorts/XXX
    58→  const ytShortsMatch = url.match(/youtube\.com\/shorts\/([a-zA-Z0-9_-]+)/);
    59→  if (ytShortsMatch) return `<iframe src="https://www.youtube.com/embed/${ytShortsMatch[1]}" width="100%" style="aspect-ratio:9/16;" frameborder="0" allowfullscreen></iframe>`;
    60→  // TikTok: tiktok.com/@user/video/XXX
    61→  const ttMatch = url.match(/tiktok\.com\/@[^/]+\/video\/(\d+)/);
    62→  if (ttMatch) return `<iframe src="https://www.tiktok.com/embed/v2/${ttMatch[1]}" width="100%" style="aspect-ratio:9/16;" frameborder="0" allowfullscreen></iframe>`;
    63→  // Direct video files (R2 uploads): .mp4, .webm, .ogg, .mov
    64→  if (/\.(mp4|webm|ogg|mov)(\?|$)/i.test(url)) {
    65→    return `<video src="${escapeHtml(url)}" controls playsinline preload="metadata" style="width:100%;border-radius:12px;background:#000;"></video>`;
    66→  }
    67→  // Other URLs: use as-is iframe
    68→  return `<iframe src="${escapeHtml(url)}" width="100%" style="aspect-ratio:9/16;" frameborder="0" allowfullscreen></iframe>`;
    69→}
